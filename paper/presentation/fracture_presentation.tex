\section{Classify and explain}

\begin{frame}{The Problem: Right, but for the Right Reason?}
  \vspace{1.2em}
  \begin{itemize}
    \item A model can be \textbf{accurate for the wrong reasons}: a metal plate, a cast or the body part can predict
          ``fractured'' better than a thin fracture line.
    \vspace{0.6em}
    \item \textbf{XAI} draws a heatmap of the evidence. It helps only if it is \textbf{faithful} (shows what the model
          uses) \textbf{and} the model uses the \textbf{fracture}.
    \vspace{0.6em}
    \item FracAtlas has the \textbf{radiologists' boxes}: we can \textbf{measure} where a model looks.
    \vspace{0.6em}
    \item Our question: can we \textbf{teach} a classifier to look at the fracture, without losing accuracy?
  \end{itemize}
\end{frame}

\begin{frame}{Six XAI Methods: How Each One Works}
  \vspace{0.3em}
  \centering\scriptsize
  \renewcommand{\arraystretch}{1.05}\setlength{\tabcolsep}{4pt}
  \begin{tabular}{>{\raggedright\arraybackslash\bfseries}p{2.5cm}>{\raggedright\arraybackslash}p{1.1cm}>{\raggedright\arraybackslash}p{5.3cm}>{\raggedright\arraybackslash}p{2.7cm}r}
    \toprule
    \textbf{method} & \textbf{family} & \textbf{how it works} & \textbf{good / bad} & \textbf{time (ms)} \\
    \midrule
    Saliency & gradient & how much the score changes if one pixel changes a little & fast / noisy & 15 \\
    Integrated Gradients & gradient & the same, averaged on the way from a black image to the X-ray & complete / slower & 194 \\
    Guided Backprop & gradient & the gradient, keeping only the positive signals & sharp / draws every edge & 10 \\
    Grad-CAM & gradient & which regions of the last feature grid push towards ``fractured'' & per class / coarse; not on ScatNet & 5 \\
    \midrule
    Occlusion & masking & cover a $16{\times}16$ square with black, see how much the score drops & honest / slow & 767 \\
    LIME & masking & hide random squares many times, fit a simple linear model & honest / slowest & 1,377 \\
    \bottomrule
  \end{tabular}\\[0.2em]
  \scriptsize time = milliseconds to explain one X-ray (ResNet18, Kaggle T4 GPU)
  \vspace{0.4em}

  \raggedright\footnotesize
  \begin{itemize}
    \item \textbf{Gradient methods}: one pass back through the network $\Rightarrow$ \textbf{fast} \mbox{(5--194\,ms)} and
          pixel-sharp. \textbf{But} they only \emph{estimate} what a tiny change would do: noisy, and in a deep network
          they can show pixels the model does not really use.
    \item \textbf{Masking methods}: hide parts of the X-ray and run the model again, hundreds of times $\Rightarrow$
          \textbf{slow} \mbox{(0.8--1.4\,s)}. \textbf{But} they \emph{measure} the real effect of removing a region: more
          faithful, though coarser ($16{\times}16$ squares).
  \end{itemize}
\end{frame}

\begin{frame}{Occlusion, Written From Scratch}
  \begin{columns}[T]
    \begin{column}{0.44\textwidth}
      \scriptsize
      \begin{algorithmic}
        \Require model $f$, X-ray $x$, class $c$, window $w{=}16$, stride $s{=}8$
        \State $z \gets f(x)_c$;\ \ $A\gets 0$;\ \ $N\gets 0$
        \For{every window position $(i,j)$, in batches}
          \State $\hat x \gets x$ with the window set to black
          \State $A[\text{window}] \mathrel{+}= z - f(\hat x)_c$
          \State $N[\text{window}] \mathrel{+}= 1$
        \EndFor
        \State \Return $A / N$ \Comment{mean drop per pixel}
      \end{algorithmic}
      \vspace{0.5em}
      \small
      \begin{itemize}
        \item Plain PyTorch, all positions scored in batches.
        \item \textbf{Checked against Captum} (the standard library): on all three models the two maps are
              \textbf{the same}. Correlation $r=1.000$; largest difference $0.0000086$, only rounding. The
              difference maps on the right are empty.
      \end{itemize}
    \end{column}
    \begin{column}{0.54\textwidth}
      \begin{figure}
        \includegraphics[width=\linewidth]{occlusion_scratch_vs_captum.png}
        \caption{Our Occlusion, Captum's, and their difference.}
      \end{figure}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Is the Map Faithful? The Deletion Test}
  \small
  We remove the parts the map calls \textbf{most important first} ($16{\times}16$ squares) and watch the model's
  confidence. \textbf{It falls fast} $\Rightarrow$ the map is right (\textbf{low area}). Grey = random order.
  \begin{figure}
    \includegraphics[width=0.92\linewidth]{deletion_curves.png}
  \end{figure}
  \vspace{-0.7em}
  \raggedright\footnotesize
  \begin{itemize}
    \item \textbf{ResNet18} starts very sure (0.95), so the drop is clear: \textbf{LIME} falls fastest (0.50), then
          Occlusion and Guided Backprop (0.56). Saliency and IG stay close to random (0.64--0.66 vs 0.70): their
          gradients saturate in a deep network.
    \item \textbf{CNN and ScatNet underfit}: they start at only $\approx$0.6, close to a guess, so there is little to
          lose and the curves stay flat. Grad-CAM is still the best for the CNN (0.50); for ScatNet every method is
          close to random.
  \end{itemize}
  \centering\small $\Rightarrow$ The best method \textbf{depends on the model}.
\end{frame}

        \item A third of the fractures are \textbf{missed} by the model.
        \item The model uses \textbf{shortcuts}: the chest, the elbow, the hand; the faithful maps honestly show them.

\partslide{II}{Teach the Model Where to Look}{One network that classifies, detects, and is trained on its own explanation}
\section{Our method}

\begin{frame}{Model A: Classifier Only (Baseline)}
  \small
  A, B and C: \textbf{same network, same data, same training}; only the \textbf{loss} changes.
  \vspace{0.2em}

  \centering
  \includegraphics[width=0.86\linewidth]{arch_model_a.png}
  \vspace{0.2em}

  \raggedright
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \footnotesize
      \textbf{What we did}: our Part I ResNet18, trained again with three changes:
      \begin{itemize}
        \item \textbf{Bigger X-rays}: 448 px instead of 224, so thin fractures stay visible.
        \item \textbf{Balanced batches}: 16 X-rays, half fractured.
        \item \textbf{Finer Grad-CAM}: a $28{\times}28$ grid instead of $7{\times}7$.
      \end{itemize}
      \textbf{Loss}: only cross-entropy (the label). Boxes come from a separate YOLO.
    \end{column}
    \begin{column}{0.46\textwidth}
      \footnotesize
      \textbf{Why it falls short}: nothing in its loss says \textbf{where}.
      \begin{itemize}
        \item Grad-CAM on the fracture: \textbf{35\%}.
        \item Blur the fracture: $p(\text{fractured})$ 81\% $\to$ \textbf{84\%}: it does not need the fracture.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Model B: Classifier + Detector}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \includegraphics[width=\linewidth]{arch_model_b.png}
    \end{column}
    \begin{column}{0.41\textwidth}
      \scriptsize
      \textbf{How it works}
      \begin{enumerate}
        \item A small \textbf{detector head} reads the \textbf{same} $28{\times}28$ grid as the classifier.
        \item For each cell it predicts: a \textbf{centre heatmap} (fracture centre here?), an \textbf{offset}
              (exact spot inside the 16-px cell) and the \textbf{box size} $(w,h)$.
        \item The radiologist's box gives the targets; the losses compare.
      \end{enumerate}
    \end{column}
  \end{columns}
  \vspace{0.2em}
  \vspace{1.2em}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \scriptsize
      \textbf{Losses} ($N$ = fracture centres, $\hat{\ }$ = predicted)\\[0.2em]
      $\mathcal{L}_{\text{focal}} = -\frac{1}{N}\sum
        \begin{cases} (1-\hat H)^2\log \hat H & \text{at a centre}\\ (1-H)^4\hat H^2\log(1-\hat H) & \text{elsewhere} \end{cases}$\\[0.3em]
      $\mathcal{L}_{\text{offset}} = \frac{1}{N}\sum |\hat o - o|$ \quad exact spot in the cell\\[0.3em]
      $\mathcal{L}_{\text{size}} = \frac{1}{N}\sum |\hat s - s|$ \quad box width and height
    \end{column}
    \begin{column}{0.49\textwidth}
      \footnotesize
      \textbf{Why it helps}: both heads share the grid, so the detector teaches it \textbf{where} fractures are:
      Grad-CAM on the fracture 35\% $\to$ \textbf{56\%}.
      \vspace{0.4em}

      \textbf{Limit}: the classifier is \emph{not forced} to use that information. Blur the fracture and its answer
      falls only from 77\% to 48\%: half of its decision still comes from elsewhere. C fixes this.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Model C: Our Joint Model (B + Explanation Losses)}
  \begin{columns}[T]
    \begin{column}{0.61\textwidth}
      \includegraphics[width=0.96\linewidth]{arch_model_c.png}
    \end{column}
    \begin{column}{0.38\textwidth}
      \scriptsize
      \textbf{How it works}: C is model B with one new idea: during training, the model also checks
      \textbf{its own explanation}.
      \vspace{0.3em}

      At every step we compute its Grad-CAM, which shows \textbf{where the classifier looks} ($p$). We then check
      three things:
      \begin{itemize}
        \item does it look \textbf{inside the radiologist's box}?
        \item does it look \textbf{where the detector looks} ($q$), and at the same place on the \textbf{mirrored}
              X-ray?
        \item if we \textbf{blur the fracture}, does the answer become ``healthy''?
      \end{itemize}
      Each ``no'' gives a loss that corrects the model.
    \end{column}
  \end{columns}
  \vspace{-0.6em}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \scriptsize\renewcommand{\arraystretch}{1.25}
      \textbf{Three extra losses} (weight 0.5, after a 2-epoch warm-up)\\[0.2em]
      \begin{tabular}{@{}l@{}}
        $\mathcal{L}_{\text{point}} = -\log \sum_{i \in \text{box}} p_i$\\
        \quad uses the \textbf{whole Grad-CAM map} and pushes it \textbf{inside the box}\\
        $\mathcal{L}_{\text{agree}} = \mathrm{JS}(p,\, q) + \mathrm{JS}\big(p,\, \mathrm{flip}(p_m)\big)$\\
        \quad must match the \textbf{detector's map} and the \textbf{mirror image}\\
        \quad JS = Jensen--Shannon divergence (0 = same map)\\
        $\mathcal{L}_{\text{erase}} = \mathrm{CE}(\tilde x_{\text{fracture}},\ \text{healthy})
          + \mathrm{CE}(\tilde x_{\text{patch}},\ \text{fractured})$\\
        \quad blur the fracture $\Rightarrow$ \textbf{healthy}; blur elsewhere $\Rightarrow$ still fractured\\
      \end{tabular}
    \end{column}
    \begin{column}{0.4\textwidth}
      \footnotesize
      \vspace*{3.2em}
      \textbf{Why it helps}: the model is trained on its \textbf{own explanation}, so it learns to look at the
      fracture: Grad-CAM on the fracture 35\% $\to$ \textbf{67\%}, with the same accuracy (93\%).
      Blur the fracture and its answer falls from 84\% to \textbf{2.5\%}: it really decides from the fracture.
    \end{column}
  \end{columns}
\end{frame}

\section{Conclusion}

\begin{frame}{Limitations and Lessons}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \textbf{Limitations}
      \begin{itemize}\small
        \item One dataset; one seed per run (two full runs of A, B, C).
        \item 104 fractured test X-rays: 4 points = 4 X-rays. Deletion test on 10 X-rays.
        \item Loss weights, thresholds, $\alpha$ set once, not tuned.
        \item The pointing game is strict: one point, a tiny box.
        \item C is trained with the blur it is tested with (controls: random patch, five untrained maps).
        \item Casts and metal plates still fool every model.
        \item The built-in detector stays far behind YOLO (AP 22--31\% vs 53\%).
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      \textbf{Lessons}
      \begin{itemize}\small
        \item \textbf{Accuracy is not enough}: 80.9\% (CNN) can mean ``finds 1 fracture in 6''.
        \item \textbf{Pretraining and resolution} beat hand-crafted invariance.
        \item A faithful map can still show a \textbf{shortcut}; the model must be \textbf{taught} to look.
        \item \textbf{Score the layer you train}, and test the decision, not only the map.
        \item A \textbf{specialist} stays best at its job: use YOLO for boxes, our model as a second opinion.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Conclusion}
  \begin{itemize}
    \item \textbf{CNN vs ScatNet}: statistically tied and close to ``never fractured''; a pretrained ResNet18 reaches
          \textbf{91.5\%}.
    \item \textbf{Six XAI methods}: the most faithful depends on the model (Grad-CAM for the CNN, LIME for ResNet18);
          our Occlusion equals Captum; but the maps point at the fracture at most \textbf{23\%} of the time.
    \item \textbf{Our joint model C} (classifier + detector + explanation losses, 448 px):
      \begin{itemize}
        \item keeps the accuracy (\textbf{93.0\%});
        \item Grad-CAM hits the fracture \textbf{67\%} of the time instead of 35\% (A); the other five improve too,
              significantly only Guided Backprop;
        \item blurring the fracture now changes the answer (84\% $\to$ \textbf{2.5\%}).
      \end{itemize}
    \item \textbf{YOLO} remains the best box finder (79\%); our model helps it by removing false boxes (AP on all X-rays
          40.7 $\to$ 46.9), and together they decide 76\% of the X-rays with \textbf{97\%} accuracy.
  \end{itemize}
\end{frame}
